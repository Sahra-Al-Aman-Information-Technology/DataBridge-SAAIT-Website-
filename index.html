  -d '{

    "auto_ddl_align": true,

from databridge import OrchestratorEngine

# Initialize Vault Authenticated DataBridge Engine
engine = OrchestratorEngine(vault_key="env.DATABRIDGE_VAULT_KEY")

# Configure Delta Sync between Oracle DB and SQL Server Target
sync_job = engine.create_pipeline(
    source_type="ORACLE_FUSION",
    target_type="SQL_SERVER",
    strategy="MERGE_UPSERT",
    tables=["GL_BALANCES", "RA_CUSTOMER_TRX_ALL"]
)

result = sync_job.execute(auto_align_ddl=True)
print(f"Sync Completed: {result.rows_inserted} rows inserted, status: {result.status}")

ALTER TABLE dw.fact_oracle_gl_balances

MERGE INTO dw.fact_oracle_gl_balances AS target
USING stg.oracle_gl_delta AS src
ON target.period_id = src.period_id AND target.account_id = src.account_id
WHEN MATCHED THEN UPDATE SET 
    target.balance_amount = src.balance_amount,
    target.last_update_date = GETDATE()
WHEN NOT MATCHED THEN INSERT (period_id, account_id, balance_amount)
VALUES (src.period_id, src.account_id, src.balance_amount);

01

Enterprise Data & IT Architecture FAQs

DataBridge supports 3 load strategies: Full Load, Append, and Merge. In Append mode, DataBridge uses a tracking column and automatically switches to Merge if today's data is already present in the target warehouse, preventing duplicates by upserting based on a Primary Key.
DataBridge registers source database stored procedures mapped to projects. Procedures are assigned sequence numbers and parallel groups—allowing independent procedures to run concurrently while dependent steps wait for prior steps to finish. Every run is logged.
DataBridge supports Oracle, SQL Server, Excel files (.xlsx / .csv), and SharePoint lists as data sources, and loads directly into Oracle and SQL Server target warehouses.
After extraction, DataBridge authenticates via Azure AD (MSAL / service principal with Client ID, Tenant ID, Client Secret stored centrally), calls the Power BI REST API, and polls until the semantic model refresh completes or fails—logging results in the Transaction Logs view.
Yes. In addition to managed cloud deployment, DataBridge is available as a Docker / Kubernetes Helm chart for self-hosted deployment inside your AWS VPC, Azure VNet, or on-premises enterprise data center.
